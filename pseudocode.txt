Module MainProgram()
    Declarations
        String routeID, peakHour, boardMore = "YES"
        Integer distanceZone, passengerCount, maxCapacity
        Real finalFare

    Output "Enter Route ID, Max Capacity, Current Count: "
    Input routeID, maxCapacity, passengerCount

    While boardMore == "YES" AND passengerCount < maxCapacity
        Output "Enter Zone (1:Short, 2:Med, 3:Long) & Peak (YES/NO): "
        Input distanceZone, peakHour

        finalFare = CalculateFare(distanceZone, peakHour)
        passengerCount = passengerCount + 1

        Output "Boarded Successfully. Fare: Le ", finalFare
        Output "Current Count: ", passengerCount, " / ", maxCapacity

        If passengerCount >= maxCapacity Then
            Output "WARNING: Vehicle Full. Boarding Closed."
        Else
            Output "Board another passenger? (YES/NO): "
            Input boardMore
        End If
    End While
End Module

Function Real CalculateFare(Integer zone, String isPeak)
    Real fare = (zone == 1) ? 5.00 : (zone == 2) ? 10.00 : 15.00
    If isPeak == "YES" Then fare = fare * 1.25
    Return fare
End Function
